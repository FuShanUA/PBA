Microsoft Access

您可以通过源探索和批量同步读取 Access 表，或者在代码中下载并解析数据库。Python 管道还可以通过替换 Azure Files 中的完整数据库文件来更新单行数据。表导出功能不可用。

Microsoft Access 连接器是一种 由 Palantir 提供的驱动程序 ，用于 Microsoft Access。

要创建新的 Microsoft Access 源，请按照 Palantir 提供的驱动程序的标准设置流程，然后使用以下部分进行 Microsoft Access 特定的配置和网络设置。有关完整的属性参考，请参阅 官方 Microsoft Access 驱动程序文档 ↗.

连接到数据库文件

Microsoft Access 由文件支撑。 ConnectionType 选择存储服务或文件访问协议，并且 DataSource 标识该 .mdb or .accdb 文件。请遵循 Microsoft Access 连接指南 ↗ 以了解属性以及 DataSource 您的存储提供商所要求的格式。

本页上的代码内源示例使用由 Azure Files 支撑的 Access 源。它们通过导入的源下载数据库，并在代码运行时中解析该文件；它们不会向代码暴露 Access API 或 JDBC 连接。

设置以下源属性：

属性标题可将排序方式切换为

对于 AccessKey 这些示例所使用的配置，请设置 AuthScheme to AccessKey 并将存储账户密钥添加为加密的 AzureAccessKey 属性。

添加一个 出口策略 以允许该源访问 <storage-account>.file.core.windows.net:443.

"""READ-only Microsoft Access table materialization through Azure Files."""

import base64
import hashlib
import hmac
import io
import os
import subprocess
import tempfile
from datetime import datetime, timezone
from urllib.parse import quote

import pandas as pd
import requests
from transforms.api import Output, transform
from transforms.external.systems import ResolvedSource, Source, external_systems

SOURCE_RID = "ri.magritte..source.<MICROSOFT_ACCESS_SOURCE_RID>"
STORAGE_ACCOUNT = "<AZURE_STORAGE_ACCOUNT>"
SHARE_NAME = "<AZURE_FILE_SHARE>"
FILE_PATH = "<PATH_TO_ACCESS_DATABASE.accdb>"
OUTPUT_PATH = (
    "/<NAMESPACE>/<PROJECT>/"
    "<FOLDER>/<OUTPUT_DATASET>"
)
AZURE_API_VERSION = "2023-11-03"
TABLE_NAMES = ("Departments", "Employees")


@external_systems(access=Source(SOURCE_RID))
@transform.using(output=Output(OUTPUT_PATH))
def read_access_tables(access: ResolvedSource, output) -> None:
    """Download, parse, and materialize Departments and Employees."""
    access_key = access.get_secret("AzureAccessKey")
    x_ms_date = datetime.now(timezone.utc).strftime("%a, %d %b %Y %H:%M:%S GMT")
    signed_headers = {
        "x-ms-date": x_ms_date,
        "x-ms-version": AZURE_API_VERSION,
    }
    canonicalized_headers = "".join(
        f"{name.lower()}:{' '.join(str(value).strip().split())}\n"
        for name, value in sorted(signed_headers.items(), key=lambda item: item[0].lower())
    )
    canonicalized_resource = f"/{STORAGE_ACCOUNT}/{SHARE_NAME}/{FILE_PATH}"
    standard_headers = ["GET", "", "", "", "", "", "", "", "", "", "", ""]
    string_to_sign = "\n".join(standard_headers) + "\n" + canonicalized_headers + canonicalized_resource
    digest = hmac.new(
        base64.b64decode(access_key),
        string_to_sign.encode("utf-8"),
        hashlib.sha256,
    ).digest()
    signature = base64.b64encode(digest).decode("ascii")
    headers = dict(signed_headers)
    headers["Authorization"] = f"SharedKey {STORAGE_ACCOUNT}:{signature}"

    encoded_path = quote(FILE_PATH, safe="/")
    url = f"https://{STORAGE_ACCOUNT}.file.core.windows.net/{SHARE_NAME}/{encoded_path}"
    proxy_uri = access.get_https_proxy_uri()
    proxies = {"http": proxy_uri, "https": proxy_uri} if proxy_uri else None

    with tempfile.TemporaryDirectory() as directory:
        database_path = os.path.join(directory, "testdb.accdb")
        response = requests.request(
            "GET",
            url,
            params=None,
            headers=headers,
            data=None,
            proxies=proxies,
            timeout=120,
            stream=True,
        )
        response.raise_for_status()
        with response:
            with open(database_path, "wb") as output_file:
                for chunk in response.iter_content(chunk_size=1024 * 1024):
                    if chunk:
                        output_file.write(chunk)

        frames = []
        for table_name in TABLE_NAMES:
            completed = subprocess.run(
                ["mdb-export", database_path, table_name],
                check=True,
                capture_output=True,
                text=True,
                timeout=120,
            )
            frame = pd.read_csv(io.StringIO(completed.stdout))
            frame.insert(0, "source_table", table_name)
            frames.append(frame)

        output.write_table(pd.concat(frames, ignore_index=True, sort=False))

def _read_table(database_path: str, table_name: str) -> list[dict[str, str | None]]:
    completed = subprocess.run(
        ["mdb-export", database_path, table_name],
        check=True,
        capture_output=True,
        text=True,
        timeout=120,
    )
    records: list[dict[str, str | None]] = []
    for row in csv.DictReader(io.StringIO(completed.stdout)):
        records.append({key: value if value != "" else None for key, value in row.items()})
    return records

def read_access_tables() -> str:
    """Return Departments and Employees as one JSON object on demand."""
    source = get_source(rid=SOURCE_RID)
    access_key = source.get_secret("AzureAccessKey")
    x_ms_date = datetime.now(timezone.utc).strftime("%a, %d %b %Y %H:%M:%S GMT")
    signed_headers = {
        "x-ms-date": x_ms_date,
        "x-ms-version": AZURE_API_VERSION,

    canonicalized_headers = "".join(
        f"{name.lower()}:{' '.join(str(value).strip().split())}\n"
        for name, value in sorted(signed_headers.items(), key=lambda item: item[0].lower())
    )
    canonicalized_resource = f"/{STORAGE_ACCOUNT}/{SHARE_NAME}/{FILE_PATH}"
    standard_headers = ["GET", "", "", "", "", "", "", "", "", "", "", ""]
    string_to_sign = "\n".join(standard_headers) + "\n" + canonicalized_headers + canonicalized_resource
    digest = hmac.new(
        base64.b64decode(access_key),
        string_to_sign.encode("utf-8"),
        hashlib.sha256,
    ).digest()
    headers = {
        **signed_headers,
        "Authorization": (
            f"SharedKey {STORAGE_ACCOUNT}:"

    response = requests.request(
        "GET",
        url,
        headers=headers,
        proxies=proxies,
        timeout=120,
        stream=True,

            payload = {
                "Departments": _read_table(database_path, "Departments"),
                "Employees": _read_table(database_path, "Employees"),
            }
    return json.dumps(payload, default=str)

import { createHmac } from "crypto";
import { Buffer } from "buffer";
import { ExternalSystems, Function, UserFacingError } from "@foundry/functions-api";
import { MicrosoftAccessSource } from "@foundry/external-systems/sources";
import MDBReader from "mdb-reader";

const STORAGE_ACCOUNT = "<AZURE_STORAGE_ACCOUNT>";
const SHARE_NAME = "<AZURE_FILE_SHARE>";
const SOURCE_FILE_PATH = "<PATH_TO_ACCESS_DATABASE.accdb>";
const AZURE_API_VERSION = "2023-11-03";

export class MicrosoftAccessReadFunctions {
    @ExternalSystems({ sources: [MicrosoftAccessSource] })
    @Function()
    public async readAccessTablesTsV1(): Promise<string> {
        const accessKey = MicrosoftAccessSource.getSecret("AzureAccessKey");
        const xMsDate = new Date().toUTCString();
        const signedHeaders: Record<string, string> = {
            "x-ms-date": xMsDate,
            "x-ms-version": AZURE_API_VERSION,
        };
        const canonicalizedHeaders = Object.entries(signedHeaders)
            .sort(([left], [right]) => left.toLowerCase().localeCompare(right.toLowerCase()))
            .map(([name, value]) => `${name.toLowerCase()}:${value.trim().replace(/\s+/g, " ")}\n`)
            .join("");
        const canonicalizedResource = `/${STORAGE_ACCOUNT}/${SHARE_NAME}/${SOURCE_FILE_PATH}`;
        const standardHeaders = ["GET", "", "", "", "", "", "", "", "", "", "", ""];
        const stringToSign = `${standardHeaders.join("\n")}\n${canonicalizedHeaders}${canonicalizedResource}`;
        const signature = createHmac("sha256", Buffer.from(accessKey, "base64"))
            .update(stringToSign, "utf8")
            .digest("base64");
        const authorization = `SharedKey ${STORAGE_ACCOUNT}:${signature}`;
        const encodedPath = SOURCE_FILE_PATH.split("/").map(encodeURIComponent).join("/");
        const fileUrl = `https://${STORAGE_ACCOUNT}.file.core.windows.net/${SHARE_NAME}/${encodedPath}`;

        const response = await MicrosoftAccessSource.fetch(fileUrl, {
            method: "GET",
            headers: {
                ...signedHeaders,
                Authorization: authorization,
            },
        });
        if (!response.ok) {
            const body = await response.text();
            throw new UserFacingError(
                `Azure Files GET ${SOURCE_FILE_PATH} failed (${response.status}): ${body.slice(0, 500)}`,
            );
        }

        const reader = new MDBReader(Buffer.from(await response.arrayBuffer()));
        const tables = {
            Departments: reader.getTable("Departments").getData(),
            Employees: reader.getTable("Employees").getData(),
        };
        return JSON.stringify(tables, (_key, item: unknown) => {
            if (typeof item === "bigint") {
                return item.toString();
            }
            if (Buffer.isBuffer(item)) {
                return item.toString("base64");
            }
            return item;
        });
    }
}

        request_headers.get(name, "")
        for name in (
            "content-encoding", "content-language", "content-length", "content-md5",
            "content-type", "date", "if-modified-since", "if-match", "if-none-match",
            "if-unmodified-since", "range",
        )
    ]
    if standard_headers[3] == "0":
        standard_headers[3] = ""
    canonicalized_headers = "".join(
        f"{name}:{' '.join(value.strip().split())}\n"
        for name, value in sorted(request_headers.items())
        if name.startswith("x-ms-")
    )
    canonicalized_resource = f"/{STORAGE_ACCOUNT}/{SHARE_NAME}/{path}"
    for name, value in sorted(request_query.items(), key=lambda item: item[0].lower()):
        canonicalized_resource += f"\n{name.lower()}:{value}"
    string_to_sign = "\n".join(standard_headers) + "\n" + canonicalized_headers + canonicalized_resource
    signature = base64.b64encode(
        hmac.new(base64.b64decode(access_key), string_to_sign.encode("utf-8"), hashlib.sha256).digest()
    ).decode("ascii")
    request_headers["Authorization"] = f"SharedKey {STORAGE_ACCOUNT}:{signature}"
    return requests.request(
        method,
        f"https://{STORAGE_ACCOUNT}.file.core.windows.net/{SHARE_NAME}/{quote(path, safe='/')}",
        params=request_query or None,
        headers=request_headers,
        data=data,
        proxies=proxies,
        timeout=120,
    )


@external_systems(access=Source(SOURCE_RID))
@transform.using(output=Output(OUTPUT_PATH))
def write_access_row(access: ResolvedSource, output) -> None:
    access_key = access.get_secret("Azure" + "Access" + "Key")
    proxy_uri = access.get_https_proxy_uri()
    proxies = {"http": proxy_uri, "https": proxy_uri} if proxy_uri else None

    with _azure_request(access_key, proxies, "GET", FILE_PATH) as response:
        if response.status_code == 404:
            output.write_table(pd.DataFrame([{"status": "skipped_target_missing", "affected_rows": 0}]))
            return
        response.raise_for_status()
        original_bytes = response.content
        original_etag = response.headers.get("ETag")
    if not original_bytes or not original_etag:
        raise RuntimeError("The target download was empty or did not return an ETag")

    with tempfile.TemporaryDirectory() as directory:
        database_path = os.path.join(directory, "working.accdb")
        with open(database_path, "wb") as database_file:
            database_file.write(original_bytes)
        library_directory = Path(__file__).resolve().parents[1] / "lib"
        connection = jaydebeapi.connect(
            "net.ucanaccess.jdbc.UcanaccessDriver",
            f"jdbc:ucanaccess://{database_path}",
            [],
            [str(library_directory / name) for name in (
                "ucanaccess-5.1.8.jar", "jackcess-5.1.7.jar", "hsqldb-2.7.4.jar",
            )],
        )
        cursor = None
        try:
            cursor = connection.cursor()
            cursor.execute(
                "UPDATE Employees SET DeptId = ? WHERE Id = ? AND Name = ? AND DeptId = ?",
                [NEW_DEPT_ID, EMPLOYEE_ID, EMPLOYEE_NAME, EXPECTED_OLD_DEPT_ID],
            )
            affected_rows = cursor.rowcount
            if affected_rows != 1:
                connection.rollback()
                raise RuntimeError(f"Expected one updated row, received {affected_rows}")
            connection.commit()
        finally:
            if cursor is not None:
                cursor.close()
            connection.close()
        with open(database_path, "rb") as database_file:
            candidate_bytes = database_file.read()

    candidate_path = f"{FILE_PATH}.candidate-{uuid.uuid4().hex}"
    lease_id = None
    replaced = False
    try:
        with _azure_request(
            access_key, proxies, "PUT", candidate_path,
            headers={"Content-Length": "0", "x-ms-content-length": str(len(candidate_bytes)), "x-ms-type": "file"},
            data=b"",
        ) as response:
            response.raise_for_status()
        for start in range(0, len(candidate_bytes), MAX_RANGE_BYTES):
            chunk = candidate_bytes[start : start + MAX_RANGE_BYTES]
            with _azure_request(
                access_key, proxies, "PUT", candidate_path,
                query={"comp": "range"},
                headers={
                    "Content-Length": str(len(chunk)),
                    "x-ms-range": f"bytes={start}-{start + len(chunk) - 1}",
                    "x-ms-write": "update",
                },
                data=chunk,
            ) as response:
                response.raise_for_status()

        with _azure_request(
            access_key, proxies, "PUT", FILE_PATH,
            query={"comp": "lease"},
            headers={"Content-Length": "0", "x-ms-lease-action": "acquire", "x-ms-lease-duration": "-1"},
            data=b"",
        ) as response:
            response.raise_for_status()
            lease_id = response.headers.get("x-ms-lease-id")
            lease_etag = response.headers.get("ETag")
        if not lease_id or lease_etag != original_etag:
            raise RuntimeError("The destination changed before replacement")

        with _azure_request(
            access_key, proxies, "PUT", FILE_PATH,
            query={"comp": "rename"},
            headers={
                "Content-Length": "0",
                "x-ms-file-rename-source": (
                    f"https://{STORAGE_ACCOUNT}.file.core.windows.net/"
                    f"{SHARE_NAME}/{quote(candidate_path, safe='/')}"
                ),
                "x-ms-file-rename-replace-if-exists": "true",
                "x-ms-destination-lease-id": lease_id,
            },
            data=b"",
        ) as response:
            response.raise_for_status()
        replaced = True
    finally:
        if lease_id is not None:
            with _azure_request(
                access_key, proxies, "PUT", FILE_PATH,
                query={"comp": "lease"},
                headers={"Content-Length": "0", "x-ms-lease-action": "release", "x-ms-lease-id": lease_id},
                data=b"",
            ) as response:
                response.raise_for_status()
        if not replaced:
            _azure_request(access_key, proxies, "DELETE", candidate_path).close()

    output.write_table(pd.DataFrame([{"status": "updated", "affected_rows": affected_rows}]))
