useTasks hook

该 useTasks hook 是一个自定义 React hook，用于管理高级待办事项应用中与特定项目相关联的任务。它利用本体论 SDK（OSDK）来获取任务数据、将其与用户信息关联，并通过订阅提供实时更新。此 hook 旨在与 stale-while-revalidate（SWR）配合使用，以实现高效的数据获取、缓存和状态管理。

此 hook 实现了实时数据订阅、批量数据检索以及结合用户信息进行高效数据增强的模式。通过在内部处理数据管理的复杂性，它为组件提供了一个简洁易用的接口来处理任务数据。

查看 useTasks 参考代码。

核心功能

  • 优化的用户数据获取： 对用户 ID 请求进行去重和批处理，以最大限度地减少 API 调用。
  • 实时更新： 实现 OSDK 订阅，使任务数据与后端保持同步。
  • SWR 集成： 针对订阅事件和手动获取，高效更新本地缓存。
  • 错误处理： 为数据获取和订阅操作提供健壮的错误处理。
  • 数据增强： 将原始 OSDK 对象转换为包含用户信息的增强型数据结构。
  • 类型安全： 通过 TypeScript 接口全程保持强类型。
  • 排序和筛选： 采用服务器端排序以优化数据呈现。
  • 依赖管理： 正确处理 useCallback 和 useEffect 钩子中的依赖项。

import React, { useEffect, useMemo, useRef, useState } from 'react';
import type { User } from '@osdk/foundry.admin';
import useTasks from '../dataServices/useTasks';
import useAdmin from '../dataServices/useAdmin';
import { IProject } from '../dataServices/useProjects';

function useProfilePictureUrls(users: User[]) {
  const { getCurrentProfilePictureUrl } = useAdmin();
  const [profilePictureUrls, setProfilePictureUrls] = useState<Record<string, string>>({});
  // Kept in a ref so the effect can check which pictures are already loaded
  // without depending on the state that it sets.
  const loadedUrls = useRef<Record<string, string>>({});

  useEffect(() => {
    let isActive = true;

    async function loadProfilePictures() {
      const usersToLoad = users.filter((user) => loadedUrls.current[user.id] === undefined);
      if (usersToLoad.length === 0) {
        return;
      }
      await Promise.all(
        usersToLoad.map(async (user) => {
          try {
            loadedUrls.current[user.id] = await getCurrentProfilePictureUrl(user);
          } catch {
            // Users without a profile picture are rendered without an avatar.
          }
        }),
      );
      if (isActive) {
        setProfilePictureUrls({ ...loadedUrls.current });
      }
    }

    loadProfilePictures();
    return () => {
      isActive = false;
    };
  }, [users, getCurrentProfilePictureUrl]);

  // `getCurrentProfilePictureUrl` returns blob URLs from `URL.createObjectURL`, which
  // the browser retains until they are revoked.
  useEffect(() => () => {
    Object.values(loadedUrls.current).forEach((url) => URL.revokeObjectURL(url));
  }, []);

  return profilePictureUrls;
}

function UserCell({ user, fallback, profilePictureUrls }: {
  user: User | undefined;
  fallback: string;
  profilePictureUrls: Record<string, string>;
}) {
  if (user == null) {
    return <div className="user-info"><span>{fallback}</span></div>;
  }

  const profilePictureUrl = profilePictureUrls[user.id];
  return (
    <div className="user-info">
      {profilePictureUrl != null && (
        <img
          src={profilePictureUrl}
          alt={`Profile picture for ${user.username}.`}
          className="user-avatar"
        />
      )}
      <span>{user.username}</span>
    </div>
  );
}

function TaskList({ project }: { project: IProject }) {
  const { tasks, isLoading, isError, metadata } = useTasks(project);
  const [filter, setFilter] = useState('ALL');

  // Deduplicate creators and assignees so each picture is requested only once,
  // in the same spirit as the batched user lookup inside the hook.
  const taskUsers = useMemo(() => {
    const usersById = new Map<string, User>();
    tasks.forEach((task) => {
      [task.createdBy, task.assignedTo].forEach((user) => {
        if (user != null) {
          usersById.set(user.id, user);
        }
      });
    });
    return Array.from(usersById.values());
  }, [tasks]);
  const profilePictureUrls = useProfilePictureUrls(taskUsers);

  if (isLoading) return <div>Loading tasks...</div>;
  if (isError) return <div>Error loading tasks: {isError.message}</div>;
  
  // Filter tasks based on the selected filter
  const filteredTasks = filter === 'ALL' 
    ? tasks 
    : tasks.filter(task => task.osdkTask.status === filter);
  
  return (
    <div className="task-list">
      <h2>Tasks for {project.name}</h2>
      
      <div className="filter-controls">
        <button 
          className={filter === 'ALL' ? 'active' : ''} 
          onClick={() => setFilter('ALL')}
        >
          All ({tasks.length})
        </button>
        <button 
          className={filter === 'COMPLETED' ? 'active' : ''} 
          onClick={() => setFilter('COMPLETED')}
        >
          Completed ({tasks.filter(t => t.osdkTask.status === 'COMPLETED').length})
        </button>
        <button 
          className={filter === 'IN PROGRESS' ? 'active' : ''} 
          onClick={() => setFilter('IN PROGRESS')}
        >
          In Progress ({tasks.filter(t => t.osdkTask.status === 'IN PROGRESS').length})
        </button>
      </div>
      
      <table className="task-table">
        <thead>
          <tr>
            <th>{metadata?.properties?.title?.displayName || 'Title'}</th>
            <th>Status</th>
            <th>Due Date</th>
            <th>Assigned To</th>
            <th>Created By</th>
          </tr>
        </thead>
        <tbody>
          {filteredTasks.map((task) => (
            <tr key={task.osdkTask.$primaryKey}>
              <td>{task.osdkTask.title}</td>
              <td>
                <span className={`status-badge ${task.osdkTask.status?.toLowerCase().replace(' ', '-') ?? 'unknown'}`}>
                  {task.osdkTask.status}
                </span>
              </td>
              <td>
                {task.osdkTask.dueDate 
                  ? new Date(task.osdkTask.dueDate).toLocaleDateString() 
                  : 'Not set'}
              </td>
              <td>
                <UserCell
                  user={task.assignedTo}
                  fallback="Unassigned"
                  profilePictureUrls={profilePictureUrls}
                />
              </td>
              <td>
                <UserCell
                  user={task.createdBy}
                  fallback="Unknown"
                  profilePictureUrls={profilePictureUrls}
                />
              </td>
            </tr>
          ))}
        </tbody>
      </table>
      
      {filteredTasks.length === 0 && (
        <div className="empty-state">
          No {filter !== 'ALL' ? filter.toLowerCase() : ''} tasks found.
        </div>
      )}
    </div>
  );
}

export default TaskList;

边缘情况与限制

在使用 useTasks hook 时，请考虑以下场景和限制：

  • 订阅的边缘情况： 订阅系统会处理缓存中已有任务的移除和更新，但该 onOutOfDate 回调不会采取任何纠正措施。新添加到对象集中的任务在下次重新验证之前也会被忽略，因为该 ADDED_OR_UPDATED 分支只会替换缓存中已存在的任务。
  • 用户详情回退： 当无法找到任务创建者或受理人的用户详情时，查找会返回 undefined ，且相应的 ITask 字段未设置。组件必须处理用户数据缺失的情况。
  • 错误恢复： 该 hook 记录订阅错误的方式是通过 onError，但 SWR fetcher 没有 try/catch。获取失败只会通过返回的 isError 值暴露出来，且重试仅限于 SWR 的内置行为。
  • 大型数据集处理： 当前实现会一次性获取所有任务而不进行分页，这在大型项目中可能会导致性能问题。
  • 复杂筛选： 所有筛选都是在获取全部任务后在客户端进行的。对于非常大的任务集，服务器端筛选会更高效。
  • 订阅竞态条件： 如果多个订阅事件在短时间内接连发生，更新缓存时就可能出现竞态条件。
  • 依赖于 useAdmin： 该 hook 所依赖的 getBatchUserDetails 来自 useAdmin ，用于解析用户信息，并且如果该请求失败，则没有健壮的回退机制。
  • 头像图片不支持批量获取： 该钩子返回的 User 对象不包含图像数据。头像图片需要为每个用户单独发送 getCurrentProfilePictureUrl 请求，如用法示例所示，并且每个请求都会返回一个 blob URL，使用方组件必须将其撤销，方法是调用 URL.revokeObjectURL()。
